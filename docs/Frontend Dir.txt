function Get-CustomTree {
    param(
        [string]$Path = ".",
        [string[]]$Exclude = @("node_modules", ".git")
    )
    
   function Show-Node ($item, $indent, $isLast) {
        if ($Exclude | Where-Object { $item.Name -like $_ }) { return }
        
        $marker = if ($isLast) { "└───" } else { "├───" }
        Write-Host "$indent$marker$($item.Name)"
        
        # Check if the item is a directory to look inside it
        if ($item.PSIsContainer) {
            $subItems = Get-ChildItem -LiteralPath $item.FullName | 
                        Where-Object { $name = $_.Name; -not ($Exclude | Where-Object { $name -like $_ }) }
            
            $padding = if ($isLast) { "    " } else { "│   " }
            $nextIndent = $indent + $padding
            
            for ($i = 0; $i -lt $subItems.Count; $i++) {
                Show-Node $subItems[$i] $nextIndent ($i -eq $subItems.Count - 1)
            }
        }
    }

    $fullPath = (Get-Item $Path).FullName
    $drive = $fullPath.Split(':')[0]
    Write-Host "Folder PATH listing for volume $drive"
    Write-Host "$($drive):."
    
    # Removed the -Directory switch so it catches root files too
    $rootItems = Get-ChildItem -Path $Path | 
                 Where-Object { $name = $_.Name; -not ($Exclude | Where-Object { $name -like $_ }) }
                 
    for ($i = 0; $i -lt $rootItems.Count; $i++) {
        Show-Node $rootItems[$i] "" ($i -eq $rootItems.Count - 1)
    }
}
